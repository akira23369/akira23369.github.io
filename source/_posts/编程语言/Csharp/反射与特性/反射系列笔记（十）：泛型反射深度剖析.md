# C# 反射系列笔记（十）：泛型反射深度剖析

> 目标：彻底理解泛型在反射层面的表示，掌握泛型类型/方法的动态构造、泛型约束检查、协变/逆变在反射中的体现，以及泛型性能优化

## 一、泛型反射的核心概念

### 泛型在运行时如何表示

```csharp
// 编译时写的代码
List<int> list = new List<int>();

// 运行时 CLR 创建的类型表示
// 泛型定义：List<T>（开放类型）
// 构造类型：List<int>（封闭类型）
```

**三个关键概念：**

| 概念 | 说明 | 示例 |
|------|------|------|
| **泛型定义 (Generic Definition)** | 带有类型参数的模板 | `typeof(List<>)` |
| **构造类型 (Constructed Type)** | 指定了具体类型参数 | `typeof(List<int>)` |
| **类型参数 (Type Parameter)** | 泛型中的 T、TKey、TValue | `typeof(List<>).GetGenericArguments()[0]` |

### 直觉理解

```
泛型定义 = 模具（形状）
    ↓
填入具体类型 = 浇铸
    ↓
构造类型 = 成品

List<T> (模具) + int (材料) = List<int> (成品)
```

### 动态构造泛型类型

```csharp
public class GenericConstructor
{
    // 基础：从泛型定义创建封闭类型
    public static Type MakeClosedType(Type openType, params Type[] typeArguments)
    {
        if (!openType.IsGenericTypeDefinition)
            throw new ArgumentException("必须是泛型定义");
        
        return openType.MakeGenericType(typeArguments);
    }
    
    // 实战：动态创建 Dictionary<string, int>
    public static object CreateDictionary()
    {
        Type openDict = typeof(Dictionary<,>);
        Type closedDict = openDict.MakeGenericType(typeof(string), typeof(int));
        return Activator.CreateInstance(closedDict);
    }
    
    // 泛型嵌套：List<Dictionary<string, int>>
    public static object CreateNestedGeneric()
    {
        Type inner = typeof(Dictionary<,>).MakeGenericType(typeof(string), typeof(int));
        Type outer = typeof(List<>).MakeGenericType(inner);
        return Activator.CreateInstance(outer);
    }
}

// 使用
var stringList = GenericConstructor.MakeClosedType(typeof(List<>), typeof(string));
object list = Activator.CreateInstance(stringList);  // new List<string>()
```

### 递归构造复杂泛型

```csharp
public static class ComplexGenericBuilder
{
    /// <summary>
    /// 根据类型字符串构造泛型类型，如 "List<int>" 或 "Dictionary<string, List<int>>"
    /// </summary>
    public static Type Parse(string typeName)
    {
        // 简化版：仅演示概念
        if (typeName == "List<int>")
            return typeof(List<int>);
        if (typeName == "Dictionary<string, List<int>>")
            return typeof(Dictionary<string, List<int>>);
        
        throw new NotSupportedException();
    }
    
    /// <summary>
    /// 递归构造：将泛型参数中的泛型也构造出来
    /// </summary>
    public static Type Build(Type genericDef, params Type[] args)
    {
        // 如果参数本身也是泛型定义，先构造它们
        var processedArgs = args.Select(arg =>
        {
            if (arg.IsGenericTypeDefinition)
                throw new ArgumentException("不能直接使用泛型定义");
            return arg;
        }).ToArray();
        
        return genericDef.MakeGenericType(processedArgs);
    }
}

// 使用
Type listOfDict = ComplexGenericBuilder.Build(
    typeof(List<>),
    typeof(Dictionary<string, int>)
);
```

### 调用泛型方法

```csharp
public class GenericMethodInvoker
{
    public static object CallGenericMethod(object target, string methodName, 
                                           Type genericArg, params object[] args)
    {
        Type targetType = target.GetType();
        
        // 1. 获取泛型方法定义
        MethodInfo genericMethod = targetType.GetMethod(methodName);
        if (genericMethod == null)
            throw new MissingMethodException($"找不到方法 {methodName}");
        
        if (!genericMethod.IsGenericMethodDefinition)
            throw new ArgumentException($"{methodName} 不是泛型方法");
        
        // 2. 构造封闭方法
        MethodInfo closedMethod = genericMethod.MakeGenericMethod(genericArg);
        
        // 3. 调用
        return closedMethod.Invoke(target, args);
    }
    
    public static T CallGenericMethod<T>(object target, string methodName, 
                                         Type genericArg, params object[] args)
    {
        return (T)CallGenericMethod(target, methodName, genericArg, args);
    }
}

// 使用
var demo = new GenericMethodDemo();
object result = GenericMethodInvoker.CallGenericMethod(demo, "Echo", typeof(string), "hello");
Console.WriteLine(result);  // "hello"

int intResult = GenericMethodInvoker.CallGenericMethod<int>(demo, "Echo", typeof(int), 42);
Console.WriteLine(intResult);  // 42
```

```csharp
public static object CallMultiGenericMethod(object target, string methodName,

{
```

## 四、泛型约束的反射验证

### 检查类型是否满足泛型约束

```csharp
public static class GenericConstraintValidator
{
    /// <summary>
    /// 检查类型是否满足泛型参数的所有约束
    /// </summary>
    /// <param name="type">待验证的实际类型（例如 typeof(int)）</param>
    /// <param name="genericParameter">泛型占位符类型（例如 typeof(List<>).GetGenericArguments()[0]）</param>
    public static bool SatisfiesConstraints(Type type, Type genericParameter)
    {
        // 验证传入的第二个参数是否真的是一个泛型参数（T），而不是具体的类型
        if (!genericParameter.IsGenericParameter)
            throw new ArgumentException("必须是泛型参数");

        // 1. 检查值类型 (where T : struct) / 引用类型 (where T : class) 约束
        // 如果约束要求是值类型，但传入的是引用类型，则不满足
        if (genericParameter.IsValueType && !type.IsValueType)
            return false;
        
        // 如果约束要求是引用类型，但传入的是值类型，则不满足
        if (genericParameter.IsClass && type.IsValueType)
            return false;

        // 2. 检查具体的接口或基类约束 (where T : BaseClass, IInterface)
        // GetGenericParameterConstraints() 会返回该泛型参数定义的所有基类和接口
        var constraints = genericParameter.GetGenericParameterConstraints();
        foreach (var constraint in constraints)
        {
            // 如果待验证类型不能分配给约束类型（即没有继承该类或实现该接口），则不满足
            if (!constraint.IsAssignableFrom(type))
                return false;
        }

        // 3. 检查无参构造函数约束 (where T : new())
        // 获取泛型参数的属性位标志
        var attributes = genericParameter.GenericParameterAttributes;
        // 通过位运算判断是否包含 DefaultConstructorConstraint 标志
        bool needDefaultCtor = (attributes & GenericParameterAttributes.DefaultConstructorConstraint) != 0;

        if (needDefaultCtor && type.IsValueType)
        {
            // 在 .NET 中，所有值类型（struct）都隐式拥有公共无参构造函数
            return true;
        }

        if (needDefaultCtor && !type.IsValueType)
        {
            // 对于引用类型，检查其是否包含公共、无参数的构造函数
            var ctor = type.GetConstructor(Type.EmptyTypes);
            if (ctor == null)
                return false;
        }

        // 经过所有检查，说明该类型符合该泛型参数的约束
        return true;
    }

    /// <summary>
    /// 获取泛型参数的所有约束描述（转为可读字符串）
    /// </summary>
    /// <param name="genericParameter">泛型占位符类型</param>
    public static string GetConstraintsDescription(Type genericParameter)
    {
        var parts = new List<string>();

        // 提取特殊约束：struct 或 class
        if (genericParameter.IsValueType)
            parts.Add("struct");
        else if (genericParameter.IsClass)
            parts.Add("class");

        // 提取具体的接口和基类约束
        foreach (var constraint in genericParameter.GetGenericParameterConstraints())
        {
            // 当标记为 struct 时，约束列表中会包含 System.ValueType，需过滤掉以保持简洁
            if (constraint == typeof(ValueType))
                continue;
            parts.Add(constraint.Name);
        }

        // 提取 new() 约束
        var attrs = genericParameter.GenericParameterAttributes;
        if ((attrs & GenericParameterAttributes.DefaultConstructorConstraint) != 0)
            parts.Add("new()");

        // 将所有约束通过逗号拼接，如果没有约束则返回 (无约束)
        return parts.Count > 0 ? string.Join(", ", parts) : "(无约束)";
    }
}

// 使用示例
public class ConstraintClass<T> where T : class, IDisposable, new() { }

Type genericParam = typeof(ConstraintClass<>).GetGenericArguments()[0];
Console.WriteLine(GenericConstraintValidator.GetConstraintsDescription(genericParam));
// 输出: class, IDisposable, new()

// 验证类型
Console.WriteLine(GenericConstraintValidator.SatisfiesConstraints(typeof(List<int>), genericParam));  // False（值类型）
Console.WriteLine(GenericConstraintValidator.SatisfiesConstraints(typeof(MemoryStream), genericParam));  // True
```

## 五、协变和逆变在反射中的体现

### 协变/逆变的基础回顾

```csharp
// 协变 (out T)：返回值类型，只能读不能写
public interface ICovariant<out T>
{
    T GetItem();
}

// 逆变 (in T)：参数类型，只能写不能读
public interface IContravariant<in T>
{
    void SetItem(T item);
}

// 不变 (T)：既可读又可写
public interface IInvariant<T>
{
    T GetItem();
    void SetItem(T item);
}
```

### 反射中检查协变/逆变

```csharp
public static void AnalyzeVariance(Type genericInterface)
{
    // 首先确保传入的类型确实是一个泛型类型（如 IEnumerable<T> 或 Action<T>）
    if (!genericInterface.IsGenericType)
        return;

    // 获取该泛型类型定义中的所有泛型参数（例如 T, TResult 等）
    var typeParams = genericInterface.GetGenericArguments();

    for (int i = 0; i < typeParams.Length; i++)
    {
        var tp = typeParams[i];
        
        // 获取泛型参数的属性位标志，这里包含了变体（Variance）信息
        var attrs = tp.GenericParameterAttributes;
        
        // 1. 检查协变：使用 'out' 关键字标记。
        // 意味着该泛型参数仅用于方法返回值（输出），允许将子类泛型赋值给父类泛型。
        bool isCovariant = (attrs & GenericParameterAttributes.Covariant) != 0;
        
        // 2. 检查逆变：使用 'in' 关键字标记。
        // 意味着该泛型参数仅用于方法参数（输入），允许将父类泛型赋值给子类泛型。
        bool isContravariant = (attrs & GenericParameterAttributes.Contravariant) != 0;
        
        // 3. 判断变体状态：
        // 如果既不是协变也不是逆变，则称为“不变（Invariant）”。
        string variance = isCovariant ? "out (协变)" : 
                         (isContravariant ? "in (逆变)" : "不变");
        
        // 输出参数的索引、名称以及它的变体类型
        Console.WriteLine($"  参数 {i}: {tp.Name} - {variance}");
    }
}

// 测试
AnalyzeVariance(typeof(IEnumerable<>));      // out T（协变）
AnalyzeVariance(typeof(Action<>));           // in T（逆变）
AnalyzeVariance(typeof(IList<>));            // 不变

/* 输出：
参数 0: T - out (协变)
参数 0: T - in (逆变)
参数 0: T - 不变
*/
```

## 六、泛型性能优化

### 缓存泛型类型构造结果

```csharp
public static class GenericTypeCache
{
    private static readonly Dictionary<string, Type> _cache = new();
    
    public static Type GetClosedType(Type openType, params Type[] args)
    {
        string key = $"{openType.FullName}<{string.Join(",", args.Select(t => t.FullName))}>";
        
        lock (_cache)
        {
            if (!_cache.TryGetValue(key, out Type closedType))
            {
                closedType = openType.MakeGenericType(args);
                _cache[key] = closedType;
            }
            return closedType;
        }
    }
}

// 使用
Type listOfInt1 = GenericTypeCache.GetClosedType(typeof(List<>), typeof(int));
Type listOfInt2 = GenericTypeCache.GetClosedType(typeof(List<>), typeof(int));
Console.WriteLine(ReferenceEquals(listOfInt1, listOfInt2));  // True
```

### 泛型方法调用缓存

```csharp
using System.Linq.Expressions;
using System.Reflection;

public class GenericMethodCache
{
    // 用于缓存编译好的委托，Key 格式为：TypeName.MethodName<T1,T2...>
    private readonly Dictionary<string, Delegate> _cache = new();

    /// <summary>
    /// 获取或创建指定泛型方法的强类型委托
    /// </summary>
    /// <param name="target">目标对象实例（如果是静态方法可为 null）</param>
    /// <param name="methodName">方法名</param>
    /// <param name="genericArgs">泛型参数数组（如 typeof(int)）</param>
    public Delegate GetOrCreateDelegate(object target, string methodName, Type[] genericArgs)
    {
        // 构造唯一的缓存键
        string key = $"{target.GetType().Name}.{methodName}<{string.Join(",", genericArgs.Select(t => t.Name))}>";
        
        lock (_cache)
        {
            if (!_cache.TryGetValue(key, out var del))
            {
                // 1. 反射获取泛型方法定义（Open Generic Method）
                var method = target.GetType().GetMethod(methodName);
                if (method == null) throw new Exception("未找到该方法");

                // 2. 绑定泛型参数，生成具体的关闭泛型方法（Closed Generic Method）
                var closedMethod = method.MakeGenericMethod(genericArgs);
                
                // 3. 将反射的 MethodInfo 编译为高性能委托
                del = CreateDynamicDelegate(closedMethod, target);
                _cache[key] = del;
            }
            return del;
        }
    }

    /// <summary>
    /// 使用表达式树动态构建：(object[] args) => (object)target.Method((ParamType)args[0], ...)
    /// </summary>
    private Delegate CreateDynamicDelegate(MethodInfo method, object target)
    {
        // 定义输入参数：一个对象数组 object[] args
        var argsParam = Expression.Parameter(typeof(object[]), "args");
        var parameters = method.GetParameters();
        
        var argExprs = new Expression[parameters.Length];
        for (int i = 0; i < parameters.Length; i++)
        {
            // 构建表达式：(ParameterType)args[i]
            // 首先从数组中取值 ArrayIndex(args, i)，然后进行类型强制转换 Convert
            argExprs[i] = Expression.Convert(
                Expression.ArrayIndex(argsParam, Expression.Constant(i)),
                parameters[i].ParameterType);
        }
        
        // 构建方法调用表达式：target.Method(arg1, arg2...)
        var callExpr = Expression.Call(
            target != null ? Expression.Constant(target) : null,
            method, argExprs);
        
        // 将返回值统一包装为 object (处理值类型装箱)
        var convertExpr = Expression.Convert(callExpr, typeof(object));
        
        // 编译为 Lambda 表达式：Func<object[], object>
        var lambda = Expression.Lambda<Func<object[], object>>(convertExpr, argsParam);
        
        return lambda.Compile();
    }
}
```

## 七、完整实战：通用仓储工厂

```csharp
// 实体基类：所有能被仓储管理的类都必须实现此接口
public interface IEntity
{
    int Id { get; set; }
}

// 泛型仓储接口：定义了标准的增删改查操作
// 约束：T 必须是类、实现 IEntity 且拥有公共无参构造函数
public interface IRepository<T> where T : class, IEntity, new()
{
    T GetById(int id);
    void Add(T entity);
    void Update(T entity);
    void Delete(int id);
}

// 具体实现：在内存中使用 Dictionary 模拟数据库存储
public class Repository<T> : IRepository<T> where T : class, IEntity, new()
{
    private readonly Dictionary<int, T> _store = new();
    private int _nextId = 1;

    public T GetById(int id) => _store.GetValueOrDefault(id);

    public void Add(T entity)
    {
        // 自动分配 ID 并存入字典
        entity.Id = _nextId++;
        _store[entity.Id] = entity;
    }

    public void Update(T entity)
    {
        if (_store.ContainsKey(entity.Id))
            _store[entity.Id] = entity;
    }

    public void Delete(int id) => _store.Remove(id);
}

// 仓储工厂：负责动态创建和管理各种实体的仓储实例
public class RepositoryFactory
{
    // 静态缓存：Key 为实体类型（如 Product），Value 为对应的 Repository<Product> 实例
    private static readonly Dictionary<Type, object> _repositories = new();

    // 获取泛型定义的 Type 对象（开放泛型类型）
    private static readonly Type _repositoryOpenType = typeof(Repository<>);
    private static readonly Type _interfaceOpenType = typeof(IRepository<>);

    /// <summary>
    /// 泛型版本：在编译时已知实体类型时使用
    /// </summary>
    public static IRepository<T> GetRepository<T>() where T : class, IEntity, new()
    {
        Type entityType = typeof(T);

        lock (_repositories) // 线程安全锁
        {
            if (!_repositories.TryGetValue(entityType, out var repo))
            {
                // 使用 MakeGenericType 将 Repository<> 转换为特定的 Repository<T>
                Type repositoryType = _repositoryOpenType.MakeGenericType(entityType);
                // 动态实例化
                repo = Activator.CreateInstance(repositoryType);
                _repositories[entityType] = repo;
            }

            return (IRepository<T>)repo;
        }
    }

    /// <summary>
    /// 运行时类型版本：当实体类型以 Type 对象形式传递时使用
    /// </summary>
    public static object GetRepository(Type entityType)
    {
        // 手动进行约束验证，防止传入未实现 IEntity 的类型
        if (!typeof(IEntity).IsAssignableFrom(entityType))
            throw new ArgumentException($"{entityType.Name} 未实现 IEntity");

        lock (_repositories)
        {
            if (!_repositories.TryGetValue(entityType, out var repo))
            {
                // 同样使用反射构造封闭泛型实例
                Type repositoryType = _repositoryOpenType.MakeGenericType(entityType);
                repo = Activator.CreateInstance(repositoryType);
                _repositories[entityType] = repo;
            }

            return repo;
        }
    }

    /// <summary>
    /// 获取仓储接口版本：确保返回的对象可以安全地转换为 IRepository<T> 对应的接口
    /// </summary>
    public static object GetRepositoryAsInterface(Type entityType)
    {
        var repo = GetRepository(entityType);
        // 构造对应的接口类型（如 IRepository<Order>）
        Type interfaceType = _interfaceOpenType.MakeGenericType(entityType);
        // 将实例显式转换为接口类型
        return Convert.ChangeType(repo, interfaceType);
    }
}

// 实体示例：产品
public class Product : IEntity
{
    public int Id { get; set; }
    public string Name { get; set; }
    public decimal Price { get; set; }
}

// 实体示例：订单
public class Order : IEntity
{
    public int Id { get; set; }
    public DateTime OrderDate { get; set; }
    public decimal Total { get; set; }
}

class Program
{
    static void Main()
    {
        // --- 场景 1：静态调用 ---
        // 编译时已知类型，直接使用泛型方法，代码提示友好
        var productRepo = RepositoryFactory.GetRepository<Product>();
        productRepo.Add(new Product { Name = "iPhone", Price = 999.99m });
        var product = productRepo.GetById(1);
        Console.WriteLine($"产品: {product.Name}");

        // --- 场景 2：动态调用 ---
        // 模拟运行时（如从字符串或配置中获取类型）
        Type orderType = typeof(Order);
        var orderRepo = RepositoryFactory.GetRepository(orderType);

        // 此时 orderRepo 是 object，需要通过反射获取方法进行调用
        var addMethod = orderRepo.GetType().GetMethod("Add");
        addMethod.Invoke(orderRepo, new object[] {
            new Order { OrderDate = DateTime.Now, Total = 100 }
        });

        Console.WriteLine("订单已通过反射添加成功。");
    }
}
```

## 八、常见陷阱与最佳实践

### 陷阱：MakeGenericType 与开放类型

```csharp
// ❌ 错误：List<int> 已经是封闭类型，不能再次构造
Type listOfInt = typeof(List<int>);
// listOfInt.MakeGenericType(typeof(string));  // ArgumentException

// ✅ 正确：从泛型定义构造
Type openList = typeof(List<>);
Type listOfString = openList.MakeGenericType(typeof(string));
```

### 陷阱：泛型参数的 GetGenericArguments 深度

```csharp
Type nested = typeof(Dictionary<string, List<int>>);

Type[] args = nested.GetGenericArguments();
Console.WriteLine(args[0]);  // System.String
Console.WriteLine(args[1]);  // System.Collections.Generic.List`1[System.Int32]

// 要获取 List<int> 的内部参数，需要继续
Type listType = args[1];
Type[] innerArgs = listType.GetGenericArguments();
Console.WriteLine(innerArgs[0]);  // System.Int32
```

### 最佳实践：泛型缓存策略

```csharp
public static class GenericCache<T>
{
    // 每个封闭泛型类型有独立的静态缓存
    public static readonly Type Type = typeof(T);
    public static readonly string Name = Type.Name;
    
    // 按 T 缓存的工厂委托
    public static Func<T> Factory { get; } = CreateFactory();
    
    private static Func<T> CreateFactory()
    {
        var ctor = typeof(T).GetConstructor(Type.EmptyTypes);
        if (ctor == null)
            return () => default;
        
        var lambda = Expression.Lambda<Func<T>>(Expression.New(ctor));
        return lambda.Compile();
    }
}

// 使用 - 零反射开销！
var product = GenericCache<Product>.Factory();
```
